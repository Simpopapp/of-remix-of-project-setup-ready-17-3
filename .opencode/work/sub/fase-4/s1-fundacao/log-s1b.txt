/**
 * Ajustes do aluno — contrato único e tipado (Fase 4, onda 0).
 *
 * Por que este módulo existe: Home, Catálogo, Ajustes, Prática e (futuramente)
 * Sessões/Simulado/Studie Plus consomem os mesmos ajustes. Se cada tela
 * inventasse o seu formato, umas apagariam as chaves das outras. Aqui vive o
 * schema, os defaults e a leitura/gravação normalizada.
 *
 * Regras de honestidade e resiliência (PRD central §4.5, §10):
 * - Nada de `localStorage` cru: a persistência é o store da Fase 3
 *   (`meta/settings`), com a mesma transação e o mesmo envelope de backup.
 * - Valor inválido NUNCA derruba a tela: cai no default daquela chave.
 * - Chaves desconhecidas são preservadas na gravação (outras fases podem
 *   adicionar seções sem que esta apague o que não conhece).
 * - `studiePlus.*` nasce desligado (PRD central §11 — pack opcional).
 * - `examDate` é chave de dia local `YYYY-MM-DD` (nunca timestamp UTC).
 *
 * Este módulo é puro quanto a I/O: só fala com o `ProgressStore` injetado, o
 * que o torna testável com `fake-indexeddb` e utilizável no servidor sem
 * tocar em `window`.

  if (typeof value !== "object" || value === null || Array.isArray(value)) return null;

  if (typeof value !== "number" || !Number.isFinite(value)) return TTS_RATE_DEFAULT;
  const clamped = Math.min(TTS_RATE_MAX, Math.max(TTS_RATE_MIN, value));
  return Math.round(clamped * 100) / 100;
}

  return value === "offline" ? "offline" : "ai";

/**
 * Normaliza qualquer entrada no schema tipado. É a fronteira de confiança:
 * o conteúdo vem de IndexedDB/backup, ou seja, é dado não confiável.
 *
 * - objeto ausente/ inválido → defaults completos;
 * - chave inválida → default daquela chave (as válidas são respeitadas);
 * - nunca lança, nunca devolve `undefined`.
 */
export function normalizeAppSettings(raw: unknown): AppSettings {
  const record = asRecord(raw);
  if (record === null) return cloneDefaults();

  const meta = asRecord(record["meta"]);
  const practice = asRecord(record["practice"]);
  const tts = asRecord(record["tts"]);
  const studiePlus = asRecord(record["studiePlus"]);

  return {
    meta: {
      targetScore: pickTargetScore(meta?.["targetScore"]),
      examDate: pickExamDate(meta?.["examDate"]),
    },
    practice: {
      timersEnabled: pickBoolean(
        practice?.["timersEnabled"],
        DEFAULT_SETTINGS.practice.timersEnabled,
      ),
      timerScale: pickTimerScale(practice?.["timerScale"]),
    },
    tts: {
      enabled: pickBoolean(tts?.["enabled"], DEFAULT_SETTINGS.tts.enabled),
      voiceUri: pickVoiceUri(tts?.["voiceUri"]),
      rate: clampSpeechRate(tts?.["rate"]),
    },
    studiePlus: {
      enabled: pickBoolean(studiePlus?.["enabled"], DEFAULT_SETTINGS.studiePlus.enabled),
      word: pickBoolean(studiePlus?.["word"], DEFAULT_SETTINGS.studiePlus.word),
      sentence: pickBoolean(studiePlus?.["sentence"], DEFAULT_SETTINGS.studiePlus.sentence),
      pronunciation: pickBoolean(
        studiePlus?.["pronunciation"],
        DEFAULT_SETTINGS.studiePlus.pronunciation,
      ),
      mode: pickMode(studiePlus?.["mode"]),
    },
  };
}

/* ------------------------------------------------------------------ */
/* Merge                                                              */
/* ------------------------------------------------------------------ */

/**
 * Merge profundo por seção (puro). Seções ausentes no patch continuam como
 * estão; o resultado é normalizado, então um patch inválido não corrompe o
 * estado — apenas não muda a chave.
 */
export function applySettingsPatch(
  current: AppSettings,
  patch: DeepPartialAppSettings | null | undefined,
): AppSettings {
  const base = normalizeAppSettings(current);
  const patchRecord = asRecord(patch);
  if (patchRecord === null) return base;

  const merged = {
    meta: { ...base.meta, ...(asRecord(patchRecord["meta"]) ?? {}) },
    practice: { ...base.practice, ...(asRecord(patchRecord["practice"]) ?? {}) },
    tts: { ...base.tts, ...(asRecord(patchRecord["tts"]) ?? {}) },
    studiePlus: { ...base.studiePlus, ...(asRecord(patchRecord["studiePlus"]) ?? {}) },
  };
  return normalizeAppSettings(merged);
}

/* ------------------------------------------------------------------ */
/* Leitura e gravação                                                 */
/* ------------------------------------------------------------------ */

/**
 * Lê os ajustes do store e devolve SEMPRE um objeto completo e válido.
 * `getSettings` devolve `{}` quando nunca houve gravação — isso vira defaults.
 */
export async function readAppSettings(store: SettingsStore): Promise<AppSettings> {
  const raw = await store.getSettings();
  return normalizeAppSettings(raw);
}

/**
 * Aplica um patch e grava. Chaves desconhecidas já existentes no store são
 * preservadas; o retorno é o estado resultante (fonte da verdade para a UI).
 */
export async function updateAppSettings(
  store: SettingsStore,
  patch: DeepPartialAppSettings,
): Promise<AppSettings> {
  const raw = await store.getSettings();
  const next = applySettingsPatch(normalizeAppSettings(raw), patch);
  // `...raw` primeiro: o schema tipado manda nas chaves que conhece, e nada
  // que outra fase tenha gravado é apagado por engano.
  await store.setSettings({ ...raw, ...next });
  return next;
}

/**
 * Cliente do progresso local — uma conexão para todo o app (Fase 4, onda 0).
 *
 * Problema que resolve: `openProgressLayer()` abre IndexedDB e monta store,
 * backup e mídia. Se cada componente abrisse a sua conexão, teríamos N conexões,
 * N caches e risco de `onblocked` entre abas. Este módulo é o singleton
 * browser-only que a UI consome.
 *
 * Regras (PRD central §4.5 e §10):
 * - O progresso é 100% local e só existe no navegador. Em SSR não há IndexedDB:
 *   `getProgressLayer()` rejeita com erro claro e `isProgressAvailable()` é
 *   `false`. Componentes só chamam dentro de `useEffect`/eventos.
 * - A Promise é a MESMA para todos os chamadores (evita corrida de abertura).
 * - Falha não envenena o singleton: a próxima chamada tenta de novo.
 * - Sem IndexedDB (navegador restrito), a UI degrada — nunca quebra a tela.
 *
 * `configureProgressClient` existe para testes (injeção de `fake-indexeddb` ou
 * `null` para simular indisponibilidade) e para uma futura troca de nome do
 * banco por implantação. Em produção nunca é chamado.
 */

import {
  ProgressDbUnavailableError,
  openProgressLayer,
  type OpenProgressLayerOptions,
  type ProgressLayer,
} from "./index";

export interface ProgressClientConfig {
  /** nome do banco; ausente = `PROGRESS_DB_NAME` da Fase 3 */
  dbName?: string;
  /** fábrica injetável; `null` força indisponibilidade */
  indexedDB?: IDBFactory | null;
}

let config: ProgressClientConfig = {};
let layerPromise: Promise<ProgressLayer> | null = null;
let resolvedLayer: ProgressLayer | null = null;

function isBrowser(): boolean {
  return typeof window !== "undefined";
}

/**
 * Há como abrir o progresso agora?
 *
 * `false` em SSR, quando não existe `indexedDB` global ou quando o cliente foi
 * configurado explicitamente como indisponível (`indexedDB: null`).
 */
export function isProgressAvailable(): boolean {
  if (!isBrowser()) return false;
  if (config.indexedDB === null) return false;
  if (config.indexedDB !== undefined) return true;
  return typeof globalThis.indexedDB !== "undefined";
}

/**
 * Abre (ou reaproveita) a conexão única do progresso.
 *
 * Rejeita com `ProgressDbUnavailableError` em SSR e quando o navegador não tem
 * IndexedDB — os consumidores tratam como estado degradado, não como crash.
 */
export function getProgressLayer(): Promise<ProgressLayer> {
  if (!isBrowser()) {
    return Promise.reject(
      new ProgressDbUnavailableError(
        "progresso local indisponível no servidor: abra dentro de useEffect/evento",
      ),
    );
  }
  if (layerPromise !== null) return layerPromise;
  if (!isProgressAvailable()) {
    return Promise.reject(
      new ProgressDbUnavailableError(
        "IndexedDB não está disponível neste navegador; o progresso não será gravado",
      ),
    );
  }

  const options: OpenProgressLayerOptions = {};
  if (config.dbName !== undefined) options.dbName = config.dbName;
  if (config.indexedDB !== undefined) options.indexedDB = config.indexedDB;

  let pending: Promise<ProgressLayer>;
  pending = openProgressLayer(options)
    .then((layer) => {
      resolvedLayer = layer;
      return layer;
    })
    .catch((error: unknown) => {
      // Não envenena o singleton: uma nova tentativa pode dar certo.
      if (layerPromise === pending) layerPromise = null;
      throw error;
    });

  layerPromise = pending;
  return pending;
}

/**
 * Acesso síncrono à camada já aberta (para handlers que não podem esperar).
 * `null` antes da primeira abertura bem-sucedida.
 */
export function peekProgressLayer(): ProgressLayer | null {
  return resolvedLayer;
}

/**
 * Fecha e esquece a conexão. Usado por testes e por um eventual "apagar tudo"
 * explícito do aluno; nenhuma migração chama isto sozinha.
 */
export async function closeProgressLayer(): Promise<void> {
  const pending = layerPromise;
  layerPromise = null;
  resolvedLayer = null;
  if (pending === null) return;
  try {
    const layer = await pending;
    layer.close();
  } catch {
    // Já estava indisponível: não há o que fechar.
  }
}

/** Reconfigura o cliente (testes/injeção). Fecha a conexão anterior se houver. */
export function configureProgressClient(next: ProgressClientConfig = {}): void {
  void closeProgressLayer();
  config = { ...next };
}

/** Volta ao estado inicial (somente testes). */
export function resetProgressClient(): void {
  void closeProgressLayer();
  config = {};
}

/**
 * Leitura em voz alta do app pela Web Speech API (Fase 4, onda 0).
 *
 * Contratos que importam (PRD central §13 e §18):
 * - O áudio fala SEMPRE o inglês original (`en-*`). Nunca "áudio da tradução"
 *   — isso é regra inviolável do Studie Plus.
 * - A UI precisa informar a voz REALMENTE usada: por isso `speak()` devolve
 *   `voiceName` (ou `null` quando não há voz em inglês).
 * - Sem voz em inglês instalada, não fingimos pronúncia: a função não fala e
 *   devolve `voiceName: null` para a tela explicar ao aluno.
 * - SSR-safe: nada de `window` em escopo de módulo. No servidor o módulo pode
 *   ser importado sem efeito; `isSpeechAvailable()` é `false`.
 *
 * O que este módulo NÃO faz: avaliar pronúncia, transcrever áudio ou tocar
 * mídia do banco. TTS de conteúdo do exame é responsabilidade dos motores.
 */

import { clampSpeechRate, type AppSettings } from "./app/settings";

/** Opções de fala. `rate` é normalizado para 0.5–2.0. */
export interface SpeakOptions {
  /** `SpeechSynthesisVoice.voiceURI`; ausente = melhor voz em inglês */
  voiceUri?: string | null;
  rate?: number;
  /** idioma desejado quando não há voz instalada; padrão `en-GB` */
  lang?: string;
  onEnd?: () => void;
  onError?: (error: SpeechSynthesisErrorEvent | Error) => void;
}

/** Handle de fala: `cancel` interrompe (inclusive o utterance anterior). */
export interface SpeakHandle {
  cancel: () => void;
  /** nome da voz realmente usada; `null` = nenhuma voz em inglês disponível */
  voiceName: string | null;
  /** `false` quando nada foi falado (SSR, sem engine ou texto vazio) */
  started: boolean;
}

export const DEFAULT_SPEECH_LANG = "en-GB";

type SpeechWindow = Window & {
  speechSynthesis?: SpeechSynthesis;
  SpeechSynthesisUtterance?: typeof SpeechSynthesisUtterance;
};

function speechWindow(): SpeechWindow | null {
  if (typeof window === "undefined") return null;
  return window as SpeechWindow;
}

function engine(): SpeechSynthesis | null {
  const win = speechWindow();
  if (win === null) return null;
  const synth = win.speechSynthesis;
  return synth ?? null;
}

/** Há engine de fala E construtor de utterance neste ambiente? */
export function isSpeechAvailable(): boolean {
  const win = speechWindow();
  if (win === null) return false;
  return win.speechSynthesis !== undefined && win.SpeechSynthesisUtterance !== undefined;
}

function isEnglishVoice(voice: SpeechSynthesisVoice): boolean {
  return voice.lang.toLowerCase().startsWith("en");
}

/**
 * Vozes em inglês instaladas, ordenadas por preferência:
 * inglês do Reino Unido, depois Estados Unidos, depois as demais variantes;
 * dentro do mesmo idioma, vozes locais primeiro (funcionam offline).
 */
export function listEnglishVoices(): SpeechSynthesisVoice[] {
  const synth = engine();
  if (synth === null) return [];
  const rank = (voice: SpeechSynthesisVoice): number => {
    const lang = voice.lang.toLowerCase();
    const langRank = lang.startsWith("en-gb") ? 0 : lang.startsWith("en-us") ? 1 : 2;
    return langRank * 2 + (voice.localService ? 0 : 1);
  };
  return synth
    .getVoices()
    .filter(isEnglishVoice)
    .slice()
    .sort((a, b) => {
      const byRank = rank(a) - rank(b);
      return byRank !== 0 ? byRank : a.name.localeCompare(b.name);
    });
}

/**
 * Assina mudanças na lista de vozes (o navegador carrega as vozes de forma
 * assíncrona). Devolve a função de cancelamento. SSR-safe (no-op).
 */
export function onVoicesChanged(handler: (voices: SpeechSynthesisVoice[]) => void): () => void {
  const synth = engine();
  if (synth === null) return () => {};
  const listener = () => handler(listEnglishVoices());
  synth.addEventListener("voiceschanged", listener);
  return () => synth.removeEventListener("voiceschanged", listener);
}

/** Escolhe a voz: URI exata quando pedida; senão a melhor voz em inglês. */
function pickVoice(synth: SpeechSynthesis, voiceUri: string | null): SpeechSynthesisVoice | null {
  const englishVoices = listEnglishVoices();
  if (voiceUri !== null) {
    const requested = synth.getVoices().find((voice) => voice.voiceURI === voiceUri);
    // Voz pedida que não é inglesa é recusada: o áudio é sempre o original.
    if (requested !== undefined && isEnglishVoice(requested)) return requested;
  }
  return englishVoices[0] ?? null;
}

/**
 * Fala um texto em inglês. Interrompe qualquer fala anterior (nunca duas
 * vozes sobrepostas). Retorna a voz usada para a UI exibir.
 */
export function speak(text: string, options: SpeakOptions = {}): SpeakHandle {
  const synth = engine();
  const win = speechWindow();
  const trimmed = text.trim();
  const Utterance = win?.SpeechSynthesisUtterance;

  if (synth === null || Utterance === undefined || trimmed.length === 0) {
    return { cancel: () => {}, voiceName: null, started: false };
  }

  const voice = pickVoice(synth, options.voiceUri ?? null);
  if (voice === null) {
    // Sem voz em inglês: não inventa pronúncia em outro idioma.
    return { cancel: () => {}, voiceName: null, started: false };
  }

  synth.cancel();

  const utterance = new Utterance(trimmed);
  utterance.voice = voice;
  utterance.lang = voice.lang || options.lang || DEFAULT_SPEECH_LANG;
  utterance.rate = clampSpeechRate(options.rate);
  if (options.onEnd !== undefined) utterance.onend = () => options.onEnd?.();
  if (options.onError !== undefined) {
    utterance.onerror = (event: SpeechSynthesisErrorEvent) => {
      // Cancelar/interromper é fluxo normal, não erro para o aluno.
      if (event.error === "canceled" || event.error === "interrupted") return;
      options.onError?.(event);
    };
  }

  synth.speak(utterance);

  return {
    cancel: () => {
      synth.cancel();
    },
    voiceName: voice.name,
    started: true,
  };
}

/** Interrompe qualquer fala em andamento (no-op no servidor). */
export function stopSpeaking(): void {
  engine()?.cancel();
}

/** Conveniência para telas: fala usando os ajustes tipados do aluno. */
export function speakWithSettings(
  text: string,
  settings: Pick<AppSettings, "tts">,
  options: Omit<SpeakOptions, "voiceUri" | "rate"> = {},
): SpeakHandle {
  return speak(text, {
    voiceUri: settings.tts.voiceUri,
    rate: settings.tts.rate,
    ...options,
  });
}

  /** revisões vencidas; `null` = ainda desconhecido */
  dueCount: number | null;
  loading: boolean;
  error: boolean;

  dueCount: number | null;
  loading: boolean;
  error: boolean;
  available: boolean;

  dueCount: null,
  loading: true,
  error: false,
  available: false,
};

export function useReviewStatus(): ReviewStatus {
  const [state, setState] = useState<ReviewState>(INITIAL_STATE);
  const [refreshToken, setRefreshToken] = useState(0);

  const refresh = useCallback(() => {
    setRefreshToken((token) => token + 1);
  }, []);

  useEffect(() => {
    let active = true;

    const load = async () => {
      if (!isProgressAvailable()) {
        if (active) {
          setState({ dueCount: null, loading: false, error: false, available: false });
        }
        return;
      }

      if (active) {
        setState((previous) => ({ ...previous, loading: true, available: true }));
      }

      try {
        const layer = await getProgressLayer();
        const due = await layer.store.getDueSnapshots(localDayKey());
        if (!active) return;
        setState({
          dueCount: due.length,
          loading: false,
          error: false,
          available: true,
        });
      } catch {
        if (!active) return;
        setState({ dueCount: null, loading: false, error: true, available: true });
      }
    };

    void load();

    const onFocus = () => {
      void load();
    };
    const onVisibilityChange = () => {
      if (document.visibilityState === "visible") void load();
    };

    window.addEventListener("focus", onFocus);
    window.addEventListener(REVIEW_STATUS_EVENT, onFocus);
    document.addEventListener("visibilitychange", onVisibilityChange);

    return () => {
      active = false;
      window.removeEventListener("focus", onFocus);
      window.removeEventListener(REVIEW_STATUS_EVENT, onFocus);
      document.removeEventListener("visibilitychange", onVisibilityChange);
    };
  }, [refreshToken]);

  return { ...state, refresh };
}

/**
 * Item de navegação com três variantes (desktop, mobile, rodapé).
 *
 * Honestidade de estado: se a rota do item ainda não existe (outra onda), o item
 * é renderizado como `span` desabilitado com "em breve" — nunca um link que
 * leva a 404 e nunca um `<a href>` que furaria o router.
 *
 * Alvos: desktop ≥ 40 px de altura, mobile ≥ 56 px (regra de 44 px do PRD §18).
 */

import { useRouterState } from "@tanstack/react-router";

import { cn } from "@/lib/utils";
import { InternalLink } from "./internal-link";
import type { NavItem } from "./nav-items";
import { isPathActive, useResolvedPath } from "./use-resolved-path";

export type NavVariant = "desktop" | "mobile" | "footer";

export interface NavLinkProps {
  item: NavItem;
  variant?: NavVariant;
  className?: string;
}

const BASE: Record<NavVariant, string> = {
  desktop:
    "inline-flex h-10 items-center gap-2 rounded-full px-3 text-sm font-medium transition-colors",
  mobile:
    "flex min-h-16 flex-1 flex-col items-center justify-center gap-1 rounded-lg px-1 text-[11px] font-medium transition-colors",
  footer: "inline-flex min-h-9 items-center rounded-md px-2 text-sm transition-colors",
};

const ACTIVE: Record<NavVariant, string> = {
  desktop: "bg-primary/10 text-primary",
  mobile: "text-primary",
  footer: "text-primary",
};

const IDLE: Record<NavVariant, string> = {
  desktop: "text-muted-foreground hover:bg-accent hover:text-foreground",
  mobile: "text-muted-foreground hover:text-foreground",
  footer: "text-muted-foreground hover:text-foreground",
};

const DISABLED: Record<NavVariant, string> = {
  desktop: "cursor-not-allowed text-muted-foreground/40",
  mobile: "cursor-not-allowed text-muted-foreground/40",
  footer: "cursor-not-allowed text-muted-foreground/40",
};

export function NavLink({ item, variant = "desktop", className }: NavLinkProps) {
  const href = useResolvedPath(item.path);
  const pathname = useRouterState({ select: (state) => state.location.pathname });
  const Icon = item.icon;
  const active = href !== null && isPathActive(pathname, href);
  const iconSize = variant === "mobile" ? "size-5" : "size-4";

  if (href === null) {
    return (
      <span
        aria-disabled="true"
        title={`${item.label} — em breve`}
        className={cn(BASE[variant], DISABLED[variant], className)}
      >
        <Icon className={iconSize} aria-hidden="true" />
        <span className="truncate">{item.label}</span>
        <span className="sr-only"> (em breve)</span>
      </span>
    );
  }

  return (
    <InternalLink
      to={href}
      title={item.description}
      aria-label={item.label}
      className={cn(BASE[variant], active ? ACTIVE[variant] : IDLE[variant], className)}
    >
      <Icon className={iconSize} aria-hidden="true" />
      <span
        className={cn(
          "truncate",
          // No cabeçalho entre md e lg só o ícone cabe; o rótulo volta em lg.
          variant === "desktop" && "hidden lg:inline",
        )}
      >
        {item.label}
      </span>
    </InternalLink>
  );
}

 * Existe um único cast (`Link` → componente de `to: string`) porque a rota de

import { Link } from "@tanstack/react-router";
import type { ComponentType, MouseEvent, ReactNode } from "react";

import { useResolvedPath } from "./use-resolved-path";

export interface InternalLinkProps {
  /** caminho desejado; resolvido para a rota registrada (com/sem barra final) */
  to: string;
  className?: string;
  children?: ReactNode;
  title?: string;
  "aria-label"?: string;
  onClick?: (event: MouseEvent<HTMLAnchorElement>) => void;
}

interface DynamicLinkProps {
  to: string;
  className?: string;
  children?: ReactNode;
  title?: string;
  "aria-label"?: string;
  onClick?: (event: MouseEvent<HTMLAnchorElement>) => void;
}

const DynamicLink = Link as unknown as ComponentType<DynamicLinkProps>;

===== src/components/layout/use-resolved-path.ts
/**
 * Resolve se um caminho de navegação existe no route tree ATUAL.
 *
 * Por que isso existe: a navegação da Fase 4 aponta para rotas que chegam em
 * ondas diferentes (Catálogo, Ajustes, Prática). Em vez de espalhar links
 * quebrados ou `<a href>` (proibido para rotas internas), o menu pergunta ao
 * router o que já está registrado: rota ausente vira item honesto "em breve",
 * rota presente vira `Link` tipado em runtime. Assim nenhum pacote precisa
 * editar o layout para "ligar" o próprio item.
 *
 * O route tree é estático (gerado no build), então o resultado é o mesmo no
 * servidor e no cliente — sem mismatch de hidratação.
 */

import { useRouter } from "@tanstack/react-router";

type RouteLike = { fullPath?: unknown } | undefined;

/**
 * Caminho registrado para `path`, tentando também a variante com barra final
 * (`/sessoes` ↔ `/sessoes/`), que é como rotas de diretório aparecem no tree.
 */
export function resolveRegisteredPath(routesByPath: unknown, path: string): string | null {
  if (typeof routesByPath !== "object" || routesByPath === null) return null;
  const routes = routesByPath as Record<string, RouteLike>;
  const candidates = path === "/" ? ["/"] : [path, `${path}/`];
  for (const candidate of candidates) {
    const route = routes[candidate];
    if (route === undefined) continue;
    const fullPath = route.fullPath;
    return typeof fullPath === "string" && fullPath.length > 0 ? fullPath : candidate;
  }
  return null;
}

/** Versão React: caminho registrado agora, ou `null` (rota ainda não existe). */
export function useResolvedPath(path: string): string | null {
  const router = useRouter();
  return resolveRegisteredPath(router.routesByPath, path);
}

/** Normaliza para comparação (barra final não muda a rota). */
export function normalizePathname(pathname: string): string {
  if (pathname.length > 1 && pathname.endsWith("/")) return pathname.slice(0, -1);
  return pathname;
}

/** `true` quando `pathname` é a rota ou uma rota filha dela. */
export function isPathActive(pathname: string, href: string): boolean {
  const current = normalizePathname(pathname);
  const target = normalizePathname(href);
  if (current === target) return true;
  if (target === "/") return false;
  return current.startsWith(`${target}/`);
}

/**
 * Abre o banco e monta os serviços com a MESMA conexão (transações atômicas
 * entre snapshot, log, agregados, backup e mídia).
 */
export async function openProgressLayer(
  options: OpenProgressLayerOptions = {},
): Promise<ProgressLayer> {
  const dbOptions: ProgressDbOpenOptions = {
    ...(options.dbName === undefined ? {} : { name: options.dbName }),
    ...(options.indexedDB === undefined ? {} : { indexedDB: options.indexedDB }),
  };
  const db = await openProgressDb(dbOptions);
  const common = {
    db,
    ...(options.now === undefined ? {} : { now: options.now }),
  };
  const store = new ProgressStore({
    ...common,
    ...(options.truncation === undefined ? {} : { truncation: options.truncation }),
    ...(options.skillWeights === undefined ? {} : { skillWeights: options.skillWeights }),
    ...(options.autoTruncate === undefined ? {} : { autoTruncate: options.autoTruncate }),
  } satisfies ProgressStoreOptions);
  const backupOptions: BackupIoOptions = {
    store,
    ...(options.now === undefined ? {} : { now: options.now }),
    ...(options.device === undefined ? {} : { device: options.device }),
    ...(options.exportAttemptLimit === undefined
      ? {}
      : { exportAttemptLimit: options.exportAttemptLimit }),
  };
  const backup = new BackupIo(backupOptions);
  const mediaOptions: MediaStoreOptions = {
    db,
    ...(options.now === undefined ? {} : { now: options.now }),
    ...(options.quotaEstimator === undefined ? {} : { quotaEstimator: options.quotaEstimator }),
  };
  const media = new MediaStore(mediaOptions);

    db,
    store,
    backup,
    media,
    close: () => {

export interface DbHarness {

  reopen: () => Promise<ProgressDatabase>;

export async function makeDb(label = "db"): Promise<DbHarness> {
  const factory = new IDBFactory();
  const name = uniqueDbName(label);
  let current = await openProgressDb({ name, indexedDB: factory });
  return {
    get db() {
      return current;
    },
    factory,
    name,
    reopen: async () => {
      current.close();
      current = await openProgressDb({ name, indexedDB: factory });
      return current;

    close: async () => {
      current.close();
      await deleteProgressDb({ name, indexedDB: factory });

  };

/**
 * Chaves de dia local — contrato de datas do produto.
 *
 * Regra do PRD central (§6.2/§9): toda data de estudo (próxima revisão do SRS, dia
 * revisado, plano do dia) é uma chave `YYYY-MM-DD` no fuso do aluno, NUNCA um
 * instante UTC (`toISOString().slice(0,10)`), porque um passo de 24 h não pode ser
 * quebrado por horário de verão nem por diferença de fuso.
 *
 * Consumidores: Fase 3 (storage/SRS), Fase 7 (fila do dia), Fase 8 (plano/relatório).
 * Este módulo é puro: zero I/O, zero dependências externas.
 */

/** `YYYY-MM-DD` no fuso local do dispositivo. Ver `localDayKey`. */
export type LocalDayKey = string;

/** Formato aceito: quatro dígitos de ano, dois de mês, dois de dia. */
export const LOCAL_DAY_KEY_PATTERN = /^(\d{4})-(\d{2})-(\d{2})$/;

function pad2(value: number): string {
  return value < 10 ? `0${value}` : String(value);
}

/**
 * Converte um `Date` na chave de dia local (`YYYY-MM-DD`) do fuso do dispositivo.
 *
 * Usa os getters locais (`getFullYear`/`getMonth`/`getDate`). NÃO use
 * `toISOString().slice(0, 10)`: em fusos a oeste de UTC isso devolve o dia anterior
 * à noite e o dia seguinte em fusos a leste nas primeiras horas.
 */
export function localDayKey(date: Date = new Date()): LocalDayKey {
  if (!(date instanceof Date) || Number.isNaN(date.getTime())) {
    throw new TypeError("localDayKey: data inválida");
  }
  return `${date.getFullYear()}-${pad2(date.getMonth() + 1)}-${pad2(date.getDate())}`;
}

/** Variante que aceita um instante ISO e devolve `null` se o instante for inválido. */
export function localDayKeyOfTimestamp(isoTimestamp: string): LocalDayKey | null {
  const parsed = Date.parse(isoTimestamp);
  if (Number.isNaN(parsed)) return null;
  return localDayKey(new Date(parsed));
}

/**
 * Valida uma chave de dia local de verdade: formato `YYYY-MM-DD` E data de
 * calendário existente (rejeita `2026-02-31`, `2026-13-01` e instantes ISO).
 */
export function isLocalDayKey(value: unknown): value is LocalDayKey {
  if (typeof value !== "string") return false;
  const match = LOCAL_DAY_KEY_PATTERN.exec(value);
  if (!match) return false;
  const year = Number(match[1]);
  const month = Number(match[2]);
  const day = Number(match[3]);
  if (month < 1 || month > 12 || day < 1 || day > 31) return false;
  // Ao meio-dia local para não colidir com transições de horário de verão.
  const probe = new Date(year, month - 1, day, 12, 0, 0, 0);
  return (
    probe.getFullYear() === year && probe.getMonth() === month - 1 && probe.getDate() === day
  );
}

/**
 * Converte a chave no `Date` do meio-dia local daquele dia.
 * Meio-dia (não meia-noite) porque em dias de transição de horário de verão a
 * meia-noite pode não existir ou ser ambígua.
 */
export function localDayKeyToDate(key: LocalDayKey): Date | null {
  if (!isLocalDayKey(key)) return null;
  const [year, month, day] = key.split("-").map(Number) as [number, number, number];
  return new Date(year, month - 1, day, 12, 0, 0, 0);
}

/**
 * Soma (ou subtrai) dias de calendário a uma chave, preservando o dia local.
 *
 * Ancorado no meio-dia local: `2026-03-07 + 1` continua `2026-03-08` mesmo num dia
 * de 23 h (início do horário de verão) e `+2` chega a `2026-03-09`.
 */
export function addLocalDays(key: LocalDayKey, days: number): LocalDayKey {
  if (!isLocalDayKey(key)) {
    throw new TypeError(`addLocalDays: chave de dia local inválida: ${String(key)}`);
  }
  if (!Number.isInteger(days)) {
    throw new TypeError(`addLocalDays: dias precisa ser inteiro, recebeu ${String(days)}`);
  }
  const [year, month, day] = key.split("-").map(Number) as [number, number, number];
  const shifted = new Date(year, month - 1, day + days, 12, 0, 0, 0);
  return localDayKey(shifted);
}
