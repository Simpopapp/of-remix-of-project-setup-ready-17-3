function deepFreeze<T>(value: T): T {
  if (typeof value === "object" && value !== null) {
    for (const nested of Object.values(value as Record<string, unknown>)) deepFreeze(nested);
    Object.freeze(value);
  }
  return value;
}

/**
 * Abre o banco e monta os serviços com a MESMA conexão (transações atômicas
 * entre snapshot, log, agregados, backup e mídia).
 */
export async function openProgressLayer(
  options: OpenProgressLayerOptions = {},
): Promise<ProgressLayer> {
  const dbOptions: ProgressDbOpenOptions = {
    ...(options.dbName === undefined ? {} : { name: options.dbName }),
    ...(options.indexedDB === undefined ? {} : { indexedDB: options.indexedDB }),
  };
  const db = await openProgressDb(dbOptions);
  const common = {
    db,
    ...(options.now === undefined ? {} : { now: options.now }),
  };
  const store = new ProgressStore({
    ...common,
    ...(options.truncation === undefined ? {} : { truncation: options.truncation }),
    ...(options.skillWeights === undefined ? {} : { skillWeights: options.skillWeights }),
    ...(options.autoTruncate === undefined ? {} : { autoTruncate: options.autoTruncate }),
  } satisfies ProgressStoreOptions);
  const backupOptions: BackupIoOptions = {
    store,
    ...(options.now === undefined ? {} : { now: options.now }),
    ...(options.device === undefined ? {} : { device: options.device }),
    ...(options.exportAttemptLimit === undefined
      ? {}
      : { exportAttemptLimit: options.exportAttemptLimit }),
  };
  const backup = new BackupIo(backupOptions);
  const mediaOptions: MediaStoreOptions = {
    db,
    ...(options.now === undefined ? {} : { now: options.now }),
    ...(options.quotaEstimator === undefined ? {} : { quotaEstimator: options.quotaEstimator }),
  };
  const media = new MediaStore(mediaOptions);

  return {
    db,
    store,
    backup,
    media,
    close: () => {
      db.close();
    },
  };
}

/**
 * Cliente do progresso local — uma conexão para todo o app (Fase 4, onda 0).
 *
 * Problema que resolve: `openProgressLayer()` abre IndexedDB e monta store,
 * backup e mídia. Se cada componente abrisse a sua conexão, teríamos N conexões,
 * N caches e risco de `onblocked` entre abas. Este módulo é o singleton
 * browser-only que a UI consome.
 *
 * Regras (PRD central §4.5 e §10):
 * - O progresso é 100% local e só existe no navegador. Em SSR não há IndexedDB:
 *   `getProgressLayer()` rejeita com erro claro e `isProgressAvailable()` é
 *   `false`. Componentes só chamam dentro de `useEffect`/eventos.
 * - A Promise é a MESMA para todos os chamadores (evita corrida de abertura).
 * - Falha não envenena o singleton: a próxima chamada tenta de novo.
 * - Sem IndexedDB (navegador restrito), a UI degrada — nunca quebra a tela.
 *
 * `configureProgressClient` existe para testes (injeção de `fake-indexeddb` ou
 * `null` para simular indisponibilidade) e para uma futura troca de nome do
 * banco por implantação. Em produção nunca é chamado.
 */

import {
  ProgressDbUnavailableError,
  openProgressLayer,
  type OpenProgressLayerOptions,
  type ProgressLayer,
} from "./index";

export interface ProgressClientConfig {
  /** nome do banco; ausente = `PROGRESS_DB_NAME` da Fase 3 */
  dbName?: string;
  /** fábrica injetável; `null` força indisponibilidade */
  indexedDB?: IDBFactory | null;
}

let config: ProgressClientConfig = {};
let layerPromise: Promise<ProgressLayer> | null = null;
let resolvedLayer: ProgressLayer | null = null;

function isBrowser(): boolean {
  return typeof window !== "undefined";
}

/**
 * Há como abrir o progresso agora?
 *
 * `false` em SSR, quando não existe `indexedDB` global ou quando o cliente foi
 * configurado explicitamente como indisponível (`indexedDB: null`).
 */
export function isProgressAvailable(): boolean {
  if (!isBrowser()) return false;
  if (config.indexedDB === null) return false;
  if (config.indexedDB !== undefined) return true;
  return typeof globalThis.indexedDB !== "undefined";
}

/**
 * Abre (ou reaproveita) a conexão única do progresso.
 *
 * Rejeita com `ProgressDbUnavailableError` em SSR e quando o navegador não tem
 * IndexedDB — os consumidores tratam como estado degradado, não como crash.
 */
export function getProgressLayer(): Promise<ProgressLayer> {
  if (!isBrowser()) {
    return Promise.reject(
      new ProgressDbUnavailableError(
        "progresso local indisponível no servidor: abra dentro de useEffect/evento",
      ),
    );
  }
  if (layerPromise !== null) return layerPromise;
  if (!isProgressAvailable()) {
    return Promise.reject(
      new ProgressDbUnavailableError(
        "IndexedDB não está disponível neste navegador; o progresso não será gravado",
      ),
    );
  }

  const options: OpenProgressLayerOptions = {};
  if (config.dbName !== undefined) options.dbName = config.dbName;
  if (config.indexedDB !== undefined) options.indexedDB = config.indexedDB;

  let pending: Promise<ProgressLayer>;
  pending = openProgressLayer(options)
    .then((layer) => {
      resolvedLayer = layer;
      return layer;
    })
    .catch((error: unknown) => {
      // Não envenena o singleton: uma nova tentativa pode dar certo.
      if (layerPromise === pending) layerPromise = null;
      throw error;
    });

  layerPromise = pending;
  return pending;
}

/**
 * Acesso síncrono à camada já aberta (para handlers que não podem esperar).
 * `null` antes da primeira abertura bem-sucedida.
 */
export function peekProgressLayer(): ProgressLayer | null {
  return resolvedLayer;
}

/**
 * Fecha e esquece a conexão. Usado por testes e por um eventual "apagar tudo"
 * explícito do aluno; nenhuma migração chama isto sozinha.
 */
export async function closeProgressLayer(): Promise<void> {
  const pending = layerPromise;
  layerPromise = null;
  resolvedLayer = null;
  if (pending === null) return;
  try {
    const layer = await pending;
    layer.close();
  } catch {
    // Já estava indisponível: não há o que fechar.
  }
}

/** Reconfigura o cliente (testes/injeção). Fecha a conexão anterior se houver. */
export function configureProgressClient(next: ProgressClientConfig = {}): void {
  void closeProgressLayer();
  config = { ...next };
}

/** Volta ao estado inicial (somente testes). */
export function resetProgressClient(): void {
  void closeProgressLayer();
  config = {};
}

/**
 * Deriva o catálogo completo a partir do manifesto do banco.
 *
 * Lança (em vez de exibir número errado) se o manifesto não cobrir os tipos esperados
 * ou se a ordem oficial não corresponder exatamente aos tipos do contrato de domínio.
 */
export function deriveCatalog(manifest: BankManifest): CatalogEntry[] {
  if (EXAM_TASK_ORDER.length !== TASK_TYPES.length) {
    invalid(
      `EXAM_TASK_ORDER tem ${EXAM_TASK_ORDER.length} entradas para ${TASK_TYPES.length} tipos`,
    );
  }

  const seen = new Set<TaskType>();
  const entries = EXAM_TASK_ORDER.map((rawType) => {
    if (!isTaskType(rawType)) invalid(`tipo desconhecido em EXAM_TASK_ORDER: ${String(rawType)}`);
    if (seen.has(rawType)) invalid(`tipo repetido em EXAM_TASK_ORDER: ${rawType}`);
    seen.add(rawType);

    const type = rawType;
    const manifestEntry = manifest.byType[type];
    if (manifestEntry === undefined) invalid(`tipo ${type} ausente do manifesto do banco`);
    if (!Number.isInteger(manifestEntry.count) || manifestEntry.count < 0) {
      invalid(`contagem inválida para ${type}: ${String(manifestEntry.count)}`);
    }
    if (!Number.isInteger(manifestEntry.words) || manifestEntry.words < 0) {
      invalid(`palavras inválidas para ${type}: ${String(manifestEntry.words)}`);
    }
    if (manifestEntry.section !== SECTION_BY_TYPE[type]) {
      invalid(`seção do manifesto (${manifestEntry.section}) não confere com o tipo ${type}`);
    }

    const entry: CatalogEntry = {
      type,
      name: TASK_DISPLAY_NAMES[type],
      section: SECTION_BY_TYPE[type],
      count: manifestEntry.count,
      words: manifestEntry.words,
      status: manifestEntry.count < SPARSE_COUNT_THRESHOLD ? "sparse" : "available",
      href: practiceHref(type),
    };
    return entry;
  });

  if (seen.size !== TASK_TYPES.length) {
    const missing = TASK_TYPES.filter((type) => !seen.has(type));
    invalid(`tipos sem posição na ordem do exame: ${missing.join(", ")}`);
  }

  return entries;
}

/**
 * Agrupa por seção preservando a ordem de aparição (que é a ordem do exame).
 * Não usa lista de seções hardcoded: se `EXAM_TASK_ORDER` mudar, o agrupamento acompanha.
 */
export function groupCatalog(entries: readonly CatalogEntry[]): CatalogSection[] {
  const buckets = new Map<Section, CatalogEntry[]>();
  for (const entry of entries) {
    const bucket = buckets.get(entry.section);
    if (bucket === undefined) buckets.set(entry.section, [entry]);
    else bucket.push(entry);
  }

  return [...buckets.entries()].map(([section, sectionEntries]) => {
    const labels = SECTION_LABELS[section];
    return {
      section,
      label: labels.label,
      labelPt: labels.labelPt,
      description: labels.description,
      entries: sectionEntries,
      types: sectionEntries.length,
      questions: sectionEntries.reduce((sum, entry) => sum + entry.count, 0),
      words: sectionEntries.reduce((sum, entry) => sum + entry.words, 0),
      sparseTypes: sectionEntries.filter((entry) => entry.status === "sparse").length,
    };
  });
}

/** Resumo do catálogo — soma das entradas, nunca uma constante. */
export function summarizeCatalog(entries: readonly CatalogEntry[]): CatalogSummary {
  return {
    types: entries.length,
    typesWithContent: entries.filter((entry) => entry.count > 0).length,
    questions: entries.reduce((sum, entry) => sum + entry.count, 0),
    words: entries.reduce((sum, entry) => sum + entry.words, 0),
    sparseTypes: entries.filter((entry) => entry.status === "sparse").length,
  };
}

 *
 * Por que isso existe: a navegação da Fase 4 aponta para rotas que chegam em
 * ondas diferentes (Catálogo, Ajustes, Prática). Em vez de espalhar links

 * rota presente vira `Link` tipado em runtime. Assim nenhum pacote precisa
 * editar o layout para "ligar" o próprio item.
 *
 * O route tree é estático (gerado no build), então o resultado é o mesmo no

import { useRouter } from "@tanstack/react-router";

type RouteLike = { fullPath?: unknown } | undefined;

/**
 * Caminho registrado para `path`, tentando também a variante com barra final
 * (`/sessoes` ↔ `/sessoes/`), que é como rotas de diretório aparecem no tree.
 */
export function resolveRegisteredPath(routesByPath: unknown, path: string): string | null {
  if (typeof routesByPath !== "object" || routesByPath === null) return null;
  const routes = routesByPath as Record<string, RouteLike>;
  const candidates = path === "/" ? ["/"] : [path, `${path}/`];
  for (const candidate of candidates) {
    const route = routes[candidate];
    if (route === undefined) continue;
    const fullPath = route.fullPath;
    return typeof fullPath === "string" && fullPath.length > 0 ? fullPath : candidate;
  }
  return null;
}

/** Versão React: caminho registrado agora, ou `null` (rota ainda não existe). */
export function useResolvedPath(path: string): string | null {
  const router = useRouter();
  return resolveRegisteredPath(router.routesByPath, path);
}

/** Normaliza para comparação (barra final não muda a rota). */
export function normalizePathname(pathname: string): string {
  if (pathname.length > 1 && pathname.endsWith("/")) return pathname.slice(0, -1);
  return pathname;
}

/** `true` quando `pathname` é a rota ou uma rota filha dela. */
export function isPathActive(pathname: string, href: string): boolean {
  const current = normalizePathname(pathname);
  const target = normalizePathname(href);
  if (current === target) return true;
  if (target === "/") return false;
  return current.startsWith(`${target}/`);
}

# Routes

TanStack Start uses **file-based routing**. Every `.tsx` file in this directory
defines a route. Do **not** create `src/pages/`, `src/routes/_app/index.tsx`, or
`app/layout.tsx` — those are Next.js / Remix conventions. The only root layout
is `src/routes/__root.tsx`.

## Conventions

| File | URL |
| --- | --- |
| `index.tsx` | `/` |
| `about.tsx` | `/about` |
| `users/index.tsx` | `/users` |
| `users/$id.tsx` | `/users/:id` (dynamic — bare `$`, no curly braces) |
| `posts/{-$category}.tsx` | `/posts/:category?` (optional segment) |
| `files/$.tsx` | `/files/*` (splat — read via `_splat` param, never `*`) |
| `_layout.tsx` | layout route (renders children via `<Outlet />`) |
| `__root.tsx` | app shell — wraps every page; preserve `<Outlet />` |

`routeTree.gen.ts` is auto-generated. Don't edit it by hand.

/**
 * Resolve se um caminho de navegação existe no route tree ATUAL.
 *
 * Por que isso existe: a navegação da Fase 4 aponta para rotas que chegam em
 * ondas diferentes (Catálogo, Ajustes, Prática). Em vez de espalhar links
 * quebrados ou `<a href>` (proibido para rotas internas), o menu pergunta ao
 * router o que já está registrado: rota ausente vira item honesto "em breve",
 * rota presente vira `Link` tipado em runtime. Assim nenhum pacote precisa
 * editar o layout para "ligar" o próprio item.
 *
 * O route tree é estático (gerado no build), então o resultado é o mesmo no
 * servidor e no cliente — sem mismatch de hidratação.
 */

import { useRouter } from "@tanstack/react-router";

type RouteLike = { fullPath?: unknown } | undefined;

/**
 * Caminho registrado para `path`, tentando também a variante com barra final
 * (`/sessoes` ↔ `/sessoes/`), que é como rotas de diretório aparecem no tree.
 */
export function resolveRegisteredPath(routesByPath: unknown, path: string): string | null {
  if (typeof routesByPath !== "object" || routesByPath === null) return null;
  const routes = routesByPath as Record<string, RouteLike>;
  const candidates = path === "/" ? ["/"] : [path, `${path}/`];
  for (const candidate of candidates) {
    const route = routes[candidate];
    if (route === undefined) continue;
    const fullPath = route.fullPath;
    return typeof fullPath === "string" && fullPath.length > 0 ? fullPath : candidate;
  }
  return null;
}

/** Versão React: caminho registrado agora, ou `null` (rota ainda não existe). */
export function useResolvedPath(path: string): string | null {
  const router = useRouter();
  return resolveRegisteredPath(router.routesByPath, path);
}

/** Normaliza para comparação (barra final não muda a rota). */
export function normalizePathname(pathname: string): string {
  if (pathname.length > 1 && pathname.endsWith("/")) return pathname.slice(0, -1);
  return pathname;
}

/** `true` quando `pathname` é a rota ou uma rota filha dela. */
export function isPathActive(pathname: string, href: string): boolean {
  const current = normalizePathname(pathname);
  const target = normalizePathname(href);
  if (current === target) return true;
  if (target === "/") return false;
  return current.startsWith(`${target}/`);
}
